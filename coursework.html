public class OlaofinLab2 {

    public static boolean bracketcheck(String input){
        Stack<Character> stack = new Stack<>();
        
        for (int i=0; i<input.length(); i++){
            char c = input.charAt(i);
            
            if (c =='(' || c == '[' || c == '{'){
                stack.push(c);
            } else if (c == ')' || c == ']' || c == '}'){
                if (stack.isEmpty()) {
                    return false;
                }
                char top = stack.top();
                
                if ((c == ')' && top != '(') || (c == ']' && top != '[') || (c == '}' && top != '{')){
                    return false;
                }
                stack.pop();
            }
    }
        return stack.isEmpty();
    }

import java.util.LinkedList;
import java.util.List;
import java.util.function.Consumer;

public class BinaryTree<A> {
    private A element;
    private BinaryTree<A> left;
    private BinaryTree<A> right;

    public BinaryTree(A element, BinaryTree<A> left, BinaryTree<A> right) {
        this.element = element;
        this.left = left;
        this.right = right;
    }

    public BinaryTree(A element) {
        this(element, null, null);
    }

    public void bfTraversal(Consumer<A> action) {
        List<BinaryTree<A>> queue = new LinkedList<>();
        queue.add(this);

        while (!queue.isEmpty()) {
            BinaryTree<A> current = queue.removeFirst();
            action.accept(current.element);

            if (current.left != null) {
                queue.add(current.left);
            }
            if (current.right != null) {
                queue.add(current.right);
            }
        }
    }

    public void preOrderTraversal(Consumer<A> action) {
        action.accept(this.element);
        
        if (this.left != null) {
            this.left.preOrderTraversal(action);
        }
        if (this.right != null) {
            this.right.preOrderTraversal(action);
        }
    }

    public void inOrderTraversal(Consumer<A> action) {
        if (this.left != null) {
            this.left.inOrderTraversal(action);
        }
        
        action.accept(this.element);
        
        if (this.right != null) {
            this.right.inOrderTraversal(action);
        }
    }

    public void postOrderTraversal(Consumer<A> action) {
        if (this.left != null) {
            this.left.postOrderTraversal(action);
        }
        if (this.right != null) {
            this.right.postOrderTraversal(action);
        }
        
        action.accept(this.element);
    }

    public int height() {
        if (this.left == null && this.right == null) {
            return 0;
        }
        
        int leftHeight = (this.left == null) ? -1 : this.left.height();
        int rightHeight = (this.right == null) ? -1 : this.right.height();
        
        return Math.max(leftHeight, rightHeight) + 1;
    }

    public boolean contains(A x) {
        if (this.element.equals(x)) {
            return true;
        }
        
        boolean inLeft = (this.left != null) && this.left.contains(x);
        boolean inRight = (this.right != null) && this.right.contains(x);
        
        return inLeft || inRight;
    }

    public static void main(String[] args) {
        BinaryTree<Character> tree =
            new BinaryTree<>('a',
                new BinaryTree<>('b', new BinaryTree<>('d'), null),
                new BinaryTree<>('c', new BinaryTree<>('e'),
                    new BinaryTree<>('f', new BinaryTree<>('g'), null)));

        System.out.print("bfTraversal: ");
        tree.bfTraversal(System.out::print);
        System.out.println();

        System.out.print("preOrderTraversal: ");
        tree.preOrderTraversal(System.out::print);
        System.out.println();

        System.out.print("inOrderTraversal: ");
        tree.inOrderTraversal(System.out::print);
        System.out.println();

        System.out.print("postOrderTraversal: ");
        tree.postOrderTraversal(System.out::print);
        System.out.println();

        System.out.println("Height: " + tree.height());
        System.out.println("contains('e'): " + tree.contains('e'));
        System.out.println("contains('h'): " + tree.contains('h'));
    }
}

import java.util.function.Consumer;


public class ThreadedTree<A extends Comparable<A>> {
    private A element;
    private ThreadedTree<A> left;
    private ThreadedTree<A> right;
    private boolean leftThread;
    private boolean rightThread;

    private ThreadedTree(A element, ThreadedTree<A> left, ThreadedTree<A> right, boolean leftThread, boolean rightThread) {
        this.element = element;
        this.left = left;
        this.right = right;
        this.leftThread = leftThread;
        this.rightThread = rightThread;
    }

    public static <A extends Comparable<A>> ThreadedTree<A> createTree(A x){
        return new ThreadedTree<>(x, null, null, true, true);
    }

    public void insert(A x){
        ThreadedTree<A> current = this;

        while(true){
            int compare = x.compareTo(current.element);

            if(compare == 0){
                return;
            } else if(compare < 0){
                if(current.leftThread){
                    break;
                }
                current = current.left;
            } else {
                if(current.rightThread){
                    break;
                }
                current = current.right;
            }
        }

        ThreadedTree<A> newNode = createTree(x);
        int compare = x.compareTo(current.element);

        if(compare < 0){
            newNode.left = current.left;
            newNode.right = current;
            current.left = newNode;
            current.leftThread = false;
        } else {
            newNode.right = current .right;
            newNode.left = current;
            current.right = newNode;
            current.rightThread = false;
        }
    }

    private ThreadedTree<A> findSuccessor(){
        if (this.rightThread){
            return this.right;
        }

        ThreadedTree<A> node = this.right;
        while(!node.leftThread){
            node = node.left;
        }
        return node;
    }

    public void inOrderTraverse(Consumer<A> action){
        ThreadedTree<A> current = this;
        while(!current.leftThread){
            current = current.left;
        }

        while( current != null){
            action.accept(current.element);
            current = current.findSuccessor();
        }
    }

    public static void main(String[] args) {
        ThreadedTree<Character> tree = ThreadedTree.createTree('h');
        tree.insert('d');
        tree.insert('b');
        tree.insert('a');
        tree.insert('c');
        tree.insert('f');
        tree.insert('e');
        tree.insert('k');
        tree.insert('i');
        tree.insert('j');

        tree.inOrderTraverse(System.out::println);
    }
}

public class AVLTree<A extends Comparable<A>> {
    private A element;
    private AVLTree<A> left;
    private AVLTree<A> right;
    private int height;

    private AVLTree(A element, AVLTree<A> left, AVLTree<A> right) {
        this.element = element;
        this.left = left;
        this.right = right;
        this.height = Math.max(height(left), height(right)) + 1;
    }

    public static <A extends Comparable<A>> AVLTree<A> createTree(A x) {
        return new AVLTree<>(x, null, null);
    }

    public static int height(AVLTree t) {
        return (t == null) ? -1 : t.height;
    }

    private AVLTree<A> rotateWithLeftChild() {
        AVLTree<A> newRootL = this.left;
        this.left = newRootL.right;
        newRootL.right = this;
        
        this.height = Math.max(height(this.left), height(this.right)) + 1;
        newRootL.height = Math.max(height(newRootL.left), height(newRootL.right)) + 1;
        return newRootL;
    }

    private AVLTree<A> rotateWithRightChild() {
        AVLTree<A> newRootR = this.right;
        this.right = newRootR.left;
        newRootR.left = this;
        
        this.height = Math.max(height(this.left), height(this.right)) + 1;
        newRootR.height = Math.max(height(newRootR.left), height(newRootR.right)) + 1;
        return newRootR;
    }

    private AVLTree<A> doubleWithLeftChild() {
        this.left = this.left.rotateWithRightChild();
        return this.rotateWithLeftChild();
    }

    private AVLTree<A> doubleWithRightChild() {
        this.right = this.right.rotateWithLeftChild();
        return this.rotateWithRightChild();
    }

    public AVLTree<A> insert(A x) {
        if (x.compareTo(this.element) < 0) {
            if (this.left == null) {
                this.left = new AVLTree<>(x, null, null);
            } else {
                this.left = this.left.insert(x);
            }
        } else if (x.compareTo(this.element) > 0) {
            if (this.right == null) {
                this.right = new AVLTree<>(x, null, null);
            } else {
                this.right = this.right.insert(x);
            }
        } else {
            return this;
        }

        this.height = Math.max(height(this.left), height(this.right)) + 1;
        int balance = height(this.left) - height(this.right);

        if (balance > 1) {
            if (x.compareTo(this.left.element) < 0) {
                return this.rotateWithLeftChild();
            } else {
                return this.doubleWithLeftChild();
            }
        }
        
        if (balance < -1) {
            if (x.compareTo(this.right.element) > 0) {
                return this.rotateWithRightChild();
            } else {
                return this.doubleWithRightChild();
            }
        }
        
        return this;
    }

    @Override
    public String toString() {
        return "AVLTree{" + "element=" + element + ", left=" + left + ", right=" + right + ", height=" + height + '}';
    }

import java.util.*;

public class Sorting {
    
    public static void radixSort(List<Integer> list){
        if (list == null || list.size() <=1){
            return;
        }
        
        int max = Collections.max(list);
        for (int place = 1; max / place > 0; place *= 10){
            List<List<Integer>> buckets = new ArrayList<>(10);
            for (int i = 0; i<10; i++){
                buckets.add(new ArrayList<>());
            }
            for (Integer num : list) {
                int digit = (num / place) % 10;
                buckets.get(digit).add(num);
            }
            list.clear();
            for ( List<Integer> bucket : buckets){
                list.addAll(bucket);
            }
        }
    }
    
    public static int partition(Integer[] a, int left, int right, int pivot){
        for(int i = left; i <= right; i++){
            if(a[i] == pivot){
                Integer temp = a[i];
                a[i] = a[right];
                a[right] = temp;
                break;
            }
        }
        int k = left - 1;
        int j = right;
        while(k<j){
            while (a[++k] < pivot);
            while (j > left && a[--j] > pivot);
            
            if(k < j){
               Integer temp = a[k];
               a[k] = a[j];
               a[j] = temp;
            }
        }
         Integer temp = a[k];
         a[k] = a[right];
         a[right] = temp;
        
        return k;
    }

        if (left < right){
            int pivot = a[right];
            int pivotIndex = partition(a, left, right, pivot);
            quickSort(a, left, pivotIndex-1);
            quickSort(a, pivotIndex+1, right);

import java.util.*;
import java.util.function.Consumer;
public class Graph<A> {
    
    private class Vertex<A> {
       A label;
       boolean wasVisited;
       
       public Vertex(A label){
           this.label = label;
           this.wasVisited = false;
       }
    }

    private Vertex<A>[] vertices;
    private int[][] adjMatrix;
    
    public Graph (A[] labels, int[][] edges){
        vertices = (Vertex<A>[]) new Vertex[labels.length];
        
        for( int i = 0; i< labels.length; i++){
            vertices[i] = new Vertex<>(labels[i]);
        }
        this.adjMatrix = edges;
    }
    
    private void resetVertices(){
        for(Vertex<A> v : vertices){
            v.wasVisited = false;
        }
    }
    
    private void dft(int vertexIndex, Consumer<A> action){
        vertices[vertexIndex].wasVisited = true;
        
        action.accept(vertices[vertexIndex].label);
        
        for (int i = 0; i< vertices.length; i++){
            if  (adjMatrix[vertexIndex][i] == 1 && !vertices[i].wasVisited){
                dft(i, action);
            }
        }
    }
    
    public void dfTraversal(int vertexIndex, Consumer<A> action){
        dft(vertexIndex, action);
        resetVertices();
    }
    
    public void bfTraversal(int vertexIndex, Consumer<A> action){
        List<Integer> queue = new LinkedList<>();
        
        vertices[vertexIndex].wasVisited = true;
        action.accept(vertices[vertexIndex].label);
        queue.add(vertexIndex);
        
        while(!queue.isEmpty()){
            int currentIndex = queue.remove(0);
            
            for(int i = 0; i< vertices.length; i++){
                if (adjMatrix[currentIndex][i] == 1 && !vertices[i].wasVisited){
                    vertices[i].wasVisited = true;
                    
                    action.accept(vertices[i].label);
                    queue.add(i);
                }
            }
        }
        resetVertices();
    }

class WeightedGraph<A> {
   private class Vertex<A> {
       A label;
       boolean wasVisited;
       int dist;
       Vertex<A> path;

       public Vertex(A label){
           this.label = label;
           this.wasVisited = false;
           this.dist = Integer.MAX_VALUE;
           this.path = null;
       }
   }

   private class PQEntry implements Comparable<PQEntry> {
       int index;
       int priority;

       public PQEntry(int index, int priority){
           this.index = index;
           this.priority = priority;
       }

       @Override
       public int compareTo(PQEntry e) {
           return Integer.compare(this.priority, e.priority);
       }
   }

    private Vertex<A>[] vertices;
    private int[][] edges;

    public WeightedGraph(A[] labels, int[][] edges){
        this.edges = edges;
        this.vertices = new Vertex[labels.length];

        for (int i = 0; i < labels.length; i++) {
            this.vertices[i] = new Vertex<>(labels[i]);
        }
    }

    private void resetVertices(){
        for (Vertex<A> v : vertices) {
            v.wasVisited = false;
            v.dist = Integer.MAX_VALUE;
            v.path = null;
        }
    }

    public Path<A> shortestPath(int startIndex, int endIndex){
        resetVertices();

        java.util.PriorityQueue<PQEntry> pq = new java.util.PriorityQueue<>();

        vertices[startIndex].dist = 0;
        pq.add(new PQEntry(startIndex, 0));

        while (!pq.isEmpty()) {
            PQEntry currentEntry = pq.remove();
            int currentIndex = currentEntry.index;
            Vertex<A> currentVertex = vertices[currentIndex];

            if (currentVertex.wasVisited) {
                continue;
            }

            currentVertex.wasVisited = true;

            if (currentIndex == endIndex) {
                break;
            }

            for (int i = 0; i < vertices.length; i++) {
                int edgeWeight = edges[currentIndex][i];

                if (edgeWeight != Integer.MAX_VALUE && !vertices[i].wasVisited) {
                    int newDist = currentVertex.dist + edgeWeight;

                    if (newDist < vertices[i].dist) {
                        vertices[i].dist = newDist;
                        vertices[i].path = currentVertex;
                        pq.add(new PQEntry(i, newDist));
                    }
                }
            }
        }
        java.util.List<A> pathList = new java.util.ArrayList<>();

        if (vertices[endIndex].dist == Integer.MAX_VALUE) {
            resetVertices();
            return new Path<>(pathList, Integer.MAX_VALUE);
        }

        Vertex<A> current = vertices[endIndex];
        while (current != null) {
            pathList.add(current.label);
            current = current.path;
        }

        java.util.Collections.reverse(pathList);

        Path<A> result = new Path<>(pathList, vertices[endIndex].dist);

        resetVertices();

        return result;
    }

    public static void main(String[] args) {
        String[] labels = { "A", "B", "C", "D", "E", "F" };
        int in = Integer.MAX_VALUE;
        int[][] edges = {
                { in,  4, in, in, in, in },
                { in, in, in,  1,  7, in },
                {  3,  2, in,  4, in, in },
                {  5, in, in, in, in, 20 },
                { in, in, in, 10, in,  6 },
                { in, in, in, in, in, in }
        };

        WeightedGraph<String> graph = new WeightedGraph<>(labels, edges);

        System.out.println(graph.shortestPath(0, 5));
        System.out.println(graph.shortestPath(2, 5));
    }
}

    var table = document.getElementById("scores");

    var tbody = table.tBodies[0];

    var rows = Array.from(tbody.rows);

    rows.sort(function(a, b) {
        return Number(a.cells[1].innerText) -
               Number(b.cells[1].innerText);

    for (var i = 0; i < rows.length; i++) {

COSC 2P89 — Assignment 2 For this assignment, you'll be adding additional functionality to your products1 website. You aren't required to build off your first assignment, but you're welcome (and encouraged) to. The informal requirements are as such: You're to have a site with multiple pages. Your site will allow for a wishlist (effectively a shopping cart, but without requiring transactions or server communication). The product overview page should have a mechanism for potentially directly adding/removing any item on the list, and each product page needs a similar mechanism for adding/removing that single item. Of course, anywhere you can change the contents of your wishlist, you must also be able to see whether the item is already in it. You are also to have an additional page, with a dynamically-generated listing of the items within your wishlist. You need to be able to remove items from it here, but not add to it. Your wishlist will be stored entirely client-side, so the server is only needed for serving HTML/CSS/JS pages. Tips Of course, you'll be using localStorage for this. First and foremost, this is a JavaScript assignment. You'll want to put some thought into precisely how you'll be storing entries. const WISHLIST_KEY = "snackbox_wishlist";

function getWishlist() {
    const raw = localStorage.getItem(WISHLIST_KEY);
    if (!raw) {
        return [];
    }
    return JSON.parse(raw);
}

function saveWishlist(list) {
    localStorage.setItem(WISHLIST_KEY, JSON.stringify(list));
}

function isInWishlist(id) {
    return getWishlist().indexOf(id) !== -1;
}

function addToWishlist(id) {
    const list = getWishlist();
    if (list.indexOf(id) === -1) {
        list.push(id);
        saveWishlist(list);
    }
}

function removeFromWishlist(id) {
    const list = getWishlist();
    saveWishlist(list.filter(item => item !== id));
}

function toggleWishlist(id) {
    if (isInWishlist(id)) {
        removeFromWishlist(id);
    } else {
        addToWishlist(id);
    }
}


function updateWishlistButton(btn, id) {
    if (isInWishlist(id)) {
        btn.textContent = "Remove from Wishlist";
        btn.classList.add("in-wishlist");
    } else {
        btn.textContent = "Add to Wishlist";
        btn.classList.remove("in-wishlist");
    }
}

function renderWishlistPage() {
    const container = document.getElementById("wishlist-container");
    if (!container) {
        return;
    }

    container.innerHTML = "";
    const list = getWishlist();

    if (list.length === 0) {
        const empty = document.createElement("p");
        empty.className = "wishlist-empty";
        empty.textContent = "Your wishlist is empty. Go to Our Snacks to add something!";
        container.appendChild(empty);
        return;
    }

    list.forEach(function (id) {
        const product = PRODUCTS[id];
        if (!product) {
            return;
        }

        const article = document.createElement("article");
        article.className = "wishlist-item";

        const heading = document.createElement("h3");
        heading.textContent = product.name;

        const desc = document.createElement("p");
        desc.textContent = product.description;

        const removeBtn = document.createElement("button");
        removeBtn.className = "wishlist-remove";
        removeBtn.textContent = "Remove";
        removeBtn.addEventListener("click", function () {
            removeFromWishlist(id);
            refreshWishlistUI();
        });

        article.appendChild(heading);
        article.appendChild(desc);
        article.appendChild(removeBtn);
        container.appendChild(article);
    });
}

function refreshWishlistUI() {
    document.querySelectorAll(".wishlist-toggle").forEach(function (btn) {
        updateWishlistButton(btn, btn.dataset.productId);
    });

    const productBtn = document.getElementById("wishlist-btn");
    if (productBtn) {
        updateWishlistButton(productBtn, productBtn.dataset.productId);
    }

    renderWishlistPage();
}

function initialize() {
    document.querySelectorAll(".wishlist-toggle").forEach(function (btn) {
        btn.addEventListener("click", function () {
            toggleWishlist(btn.dataset.productId);
            refreshWishlistUI();
        });
    });

    const productBtn = document.getElementById("wishlist-btn");
    if (productBtn) {
        productBtn.addEventListener("click", function () {
            toggleWishlist(productBtn.dataset.productId);
            refreshWishlistUI();
        });
    }

    refreshWishlistUI();
}

window.addEventListener("load", initialize, false);

const WISHLIST_KEY = "snackbox_wishlist";

const loggedIn = document.body.dataset.loggedIn == "yes";

let serverWishlist = [];

// Get the local wishlist
function getLocalWishlist() {

    try {
        const list = JSON.parse(localStorage.getItem(WISHLIST_KEY) || "[]");

        if (Array.isArray(list)) {
            return list;
        }
    } catch (error) {
        return [];
    }

    return [];
}

// Save the local wishlist
function saveLocalWishlist(list) {
    localStorage.setItem(WISHLIST_KEY, JSON.stringify(list));
}

// Send a request to the PHP script
function sendRequest(action, id, ids, callback) {

    const request = new XMLHttpRequest();

    request.open("POST", "wishlist-api.php", true);

    request.setRequestHeader("Content-Type", "application/json");

    request.onreadystatechange = function () {

        if (request.readyState == 4) {

            if (request.status == 200) {

                try {
                    const list = JSON.parse(request.responseText);

                    if (Array.isArray(list)) {
                        callback(list);
                        return;
                    }
                } catch (error) {
                    // Invalid server response
                }
            }

            alert("Wishlist could not be updated.");
        }
    };

    request.send(JSON.stringify({
        action: action,
        id: id,
        ids: ids
    }));
}

// Get the current wishlist
function getWishlist() {

    if (loggedIn) {
        return serverWishlist;
    }

    return getLocalWishlist();
}

// Add or remove a product
function toggleWishlist(id) {

    const list = getWishlist();

    if (loggedIn) {

        let action = "add";

        if (list.indexOf(id) !== -1) {
            action = "remove";
        }

        sendRequest(action, id, [], function (newList) {
            serverWishlist = newList;
            updatePage();
        });

    } else {

        if (list.indexOf(id) !== -1) {
            saveLocalWishlist(list.filter(item => item !== id));
        } else {
            list.push(id);
            saveLocalWishlist(list);
        }

        updatePage();
    }
}

// Remove an item from the wishlist page
function removeWishlistItem(id) {

    if (loggedIn) {

        sendRequest("remove", id, [], function (newList) {
            serverWishlist = newList;
            updatePage();
        });

    } else {

        const list = getLocalWishlist();

        saveLocalWishlist(list.filter(item => item !== id));

        updatePage();
    }
}

// Update buttons and visible wishlist items
function updatePage() {

    const list = getWishlist();

    document.querySelectorAll(".wishlist-toggle").forEach(function (button) {

        const id = button.dataset.productId;

        if (list.indexOf(id) !== -1) {
            button.textContent = "Remove from Wishlist";
            button.classList.add("in-wishlist");
        } else {
            button.textContent = "Add to Wishlist";
            button.classList.remove("in-wishlist");
        }
    });

    let count = 0;

    document.querySelectorAll(".wishlist-item").forEach(function (item) {

        const id = item.dataset.productId;

        if (list.indexOf(id) !== -1) {
            item.style.display = "block";
            count++;
        } else {
            item.style.display = "none";
        }
    });

    const empty = document.getElementById("wishlist-empty");

    if (empty) {
        empty.style.display = count == 0 ? "block" : "none";
    }
}

// Start the wishlist
function initializeWishlist() {

    document.querySelectorAll(".wishlist-toggle").forEach(function (button) {

        button.addEventListener("click", function () {
            toggleWishlist(button.dataset.productId);
        });
    });

    document.querySelectorAll(".wishlist-remove").forEach(function (button) {

        button.addEventListener("click", function () {
            removeWishlistItem(button.dataset.productId);
        });
    });

    if (loggedIn) {

        // Merge local items with server items after login
        const localList = getLocalWishlist();

        sendRequest("merge", "", localList, function (newList) {

            serverWishlist = newList;

            localStorage.removeItem(WISHLIST_KEY);

            updatePage();
        });

    } else {

        updatePage();
    }
}

if (!isset($_SESSION["username"])) {

// Location of the saved wishlist
$file = __DIR__ . "/data/wishlist.json";

// Read the saved wishlist
$wishlist = json_decode(file_get_contents($file), true);

if (!is_array($wishlist)) {
    $wishlist = array();
}

// Check that a product actually exists
function validProduct($id) {

    if (!is_string($id)) {
        return false;
    }

    if (!preg_match("/^[a-z0-9_-]+$/", $id)) {
        return false;
    }

    return file_exists(__DIR__ . "/products/" . $id . ".json");
}

// Remove invalid IDs and duplicates
$clean = array();

foreach ($wishlist as $id) {
    if (validProduct($id) && !in_array($id, $clean)) {
        $clean[] = $id;
    }
}

$wishlist = $clean;

// Read the AJAX request
$data = json_decode(file_get_contents("php://input"), true);

$action = $data["action"] ?? "";
$id = $data["id"] ?? "";

// Add a product
if ($action == "add") {

    if (validProduct($id) && !in_array($id, $wishlist)) {
        $wishlist[] = $id;
    }

// Remove a product
} elseif ($action == "remove") {

    $wishlist = array_values(array_diff($wishlist, array($id)));

// Merge the local wishlist with the server wishlist
} elseif ($action == "merge") {

    $ids = $data["ids"] ?? array();

    if (is_array($ids)) {

        foreach ($ids as $item) {

            if (validProduct($item) && !in_array($item, $wishlist)) {
                $wishlist[] = $item;
            }
        }
    }

} else {

    http_response_code(400);
    echo json_encode(array());
    exit;
}

// Save the updated wishlist
if (file_put_contents($file, json_encode($wishlist), LOCK_EX) === false) {
    http_response_code(500);
    echo json_encode(array());
    exit;
}

// Send the updated wishlist back to JavaScript
echo json_encode($wishlist);
